def rss_bytes() -> int:
    if platform.system() == "Windows":
        import ctypes
        from ctypes import wintypes

        class Counters(ctypes.Structure):
            _fields_ = [
                ("cb", wintypes.DWORD),
                ("PageFaultCount", wintypes.DWORD),
                ("PeakWorkingSetSize", ctypes.c_size_t),
                ("WorkingSetSize", ctypes.c_size_t),
                ("QuotaPeakPagedPoolUsage", ctypes.c_size_t),
                ("QuotaPagedPoolUsage", ctypes.c_size_t),
                ("QuotaPeakNonPagedPoolUsage", ctypes.c_size_t),
                ("QuotaNonPagedPoolUsage", ctypes.c_size_t),
                ("PagefileUsage", ctypes.c_size_t),
                ("PeakPagefileUsage", ctypes.c_size_t),
            ]

        counters = Counters()
        counters.cb = ctypes.sizeof(Counters)
        ctypes.windll.psapi.GetProcessMemoryInfo(
            ctypes.windll.kernel32.GetCurrentProcess(), ctypes.byref(counters), counters.cb
        )
        return int(counters.PeakWorkingSetSize)
    import resource

    return resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024


def grant_from_record(heading: str) -> authorization.ReadGrant:
    import re

    text = RECORD.read_text(encoding="utf-8")
    section = re.split(
        r"^## ", text.split(heading, 1)[1], maxsplit=1, flags=re.MULTILINE
    )[0]
    row = {
        f: v.strip().strip("`")
        for f, v in re.findall(r"^\| \*\*(\w+)\*\* \| (.+?) \|$", section, re.MULTILINE)
    }
    return authorization.ReadGrant(
        operation=row["operation"],
        span_start_utc=row["span_start_utc"],
        span_end_utc=row["span_end_utc"],
        pairs=tuple(row["pairs_explicit"].split()),
        timeframe=row["timeframe"],
        approved_head_sha=row["approved_head_sha"],
        approved_implementation_fingerprint=row["approved_implementation_fingerprint"],
        approver_record=row["approver_record"],
    )


def log(message: str) -> None:
    stamp = datetime.now(UTC).strftime("%H:%M:%S")
    print(f"[{stamp}] {message}", flush=True)

def safely(what, thunk, default=None):
    """A reporting bug must not lose the outcome of an irreversible run."""
    try:
        return thunk()
    except Exception as exc:  # noqa: BLE001 - reporting only
        log(f"post-run report: {what} unavailable ({type(exc).__name__}: {exc})")
        return default
